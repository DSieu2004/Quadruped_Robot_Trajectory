cmake_minimum_required(VERSION 3.22)
project(go2_smc_ros VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(OpenGL REQUIRED)
find_package(glfw3 3.3 REQUIRED)
find_package(Eigen3 3.3 REQUIRED NO_MODULE)
find_package(rclcpp REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(rosgraph_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(nav2_msgs REQUIRED)
find_package(nav2_core REQUIRED)
find_package(nav2_costmap_2d REQUIRED)
find_package(pluginlib REQUIRED)
find_package(tf2_geometry_msgs REQUIRED)
find_package(lifecycle_msgs REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(TINYXML2 REQUIRED IMPORTED_TARGET tinyxml2)
get_filename_component(GO2_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_subdirectory("${GO2_ROOT}" "${CMAKE_CURRENT_BINARY_DIR}/core")

add_executable(mujoco_bridge src/mujoco_bridge.cpp)
target_link_libraries(mujoco_bridge quadruped::go2_control glfw OpenGL::GL)
ament_target_dependencies(mujoco_bridge rclcpp sensor_msgs geometry_msgs nav_msgs rosgraph_msgs std_msgs tf2_ros)
target_compile_options(mujoco_bridge PRIVATE -Wall -Wextra -Wpedantic)
target_compile_definitions(mujoco_bridge PRIVATE GO2_SCENE="${GO2_ROOT}/models/quadruped/slam_scene.xml")

add_executable(description_generator src/description_generator.cpp)
target_link_libraries(description_generator PRIVATE mujoco_sdk PkgConfig::TINYXML2)
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/urdf")
add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/urdf/go2.urdf"
  COMMAND description_generator "${GO2_ROOT}/models/quadruped/go2.xml" "${CMAKE_CURRENT_BINARY_DIR}/urdf/go2.urdf"
  DEPENDS description_generator "${GO2_ROOT}/models/quadruped/go2.xml")
add_custom_target(go2_description ALL DEPENDS "${CMAKE_CURRENT_BINARY_DIR}/urdf/go2.urdf")

add_executable(mapping_driver src/mapping_driver.cpp)
ament_target_dependencies(mapping_driver rclcpp geometry_msgs nav_msgs std_msgs)
add_executable(navigation_goal src/navigation_goal.cpp)
ament_target_dependencies(navigation_goal rclcpp rclcpp_action geometry_msgs nav2_msgs std_msgs lifecycle_msgs)
add_executable(trajectory_follower "${GO2_ROOT}/src/trajectory/trajectory_follower.cpp")
target_link_libraries(trajectory_follower quadruped::smc quadruped::pid quadruped::mpc)
ament_target_dependencies(trajectory_follower rclcpp geometry_msgs nav_msgs std_msgs)
add_library(go2_path_controller SHARED src/smc_path_controller.cpp)
target_link_libraries(go2_path_controller quadruped::smc quadruped::pid)
ament_target_dependencies(go2_path_controller rclcpp nav2_core nav2_costmap_2d pluginlib tf2_geometry_msgs)
pluginlib_export_plugin_description_file(nav2_core controller_plugins.xml)
install(TARGETS go2_path_controller LIBRARY DESTINATION lib)
install(TARGETS mujoco_bridge mapping_driver navigation_goal trajectory_follower DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS
  "${GO2_ROOT}/src/trajectory/trajectory_plotter.py"
  "${GO2_ROOT}/src/trajectory/trajectory_compare.py"
  DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
install(DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/urdf" DESTINATION share/${PROJECT_NAME})
install(DIRECTORY "${GO2_ROOT}/models/quadruped/assets/" DESTINATION share/${PROJECT_NAME}/meshes)
install(FILES "${GO2_ROOT}/models/quadruped/LICENSE" DESTINATION share/${PROJECT_NAME}/meshes)
ament_package()
